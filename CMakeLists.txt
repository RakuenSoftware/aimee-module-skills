cmake_minimum_required(VERSION 3.16)
project(aimee_module_skills VERSION 0.4.0 LANGUAGES C)

include(GNUInstallDirs)
find_package(aimee-core 0.4.0 EXACT CONFIG REQUIRED)
if(NOT TARGET aimee::aimee-core-event-bus-client)
    message(FATAL_ERROR "aimee-module-skills requires the Linux event-bus client")
endif()
add_executable(aimee-module-skills runtime/main.c src/modules/skills/module_adapter.c)
target_compile_features(aimee-module-skills PRIVATE c_std_11)
target_include_directories(aimee-module-skills PRIVATE src/modules/skills/include)
target_link_libraries(aimee-module-skills PRIVATE aimee::aimee-core-event-bus-client)
configure_file(grants/module.grant.in ${CMAKE_CURRENT_BINARY_DIR}/skills.grant @ONLY)
install(TARGETS aimee-module-skills RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/skills.grant
        DESTINATION ${CMAKE_INSTALL_DATADIR}/aimee/module-grants)
